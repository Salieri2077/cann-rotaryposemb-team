cmake_minimum_required(VERSION 3.16)
find_package(ASC REQUIRED)
project(rotary_pos_emb_validation LANGUAGES ASC CXX)
set(CMAKE_CXX_STANDARD 14)
if(DEFINED NPU_ARCH)
    set(SOC_ARCH ${NPU_ARCH})
else()
    set(SOC_ARCH "dav-2201")
endif()
add_executable(rope_npu_runner npu_runner.asc)
target_link_libraries(rope_npu_runner PRIVATE
    tiling_api register platform unified_dlog dl m graph_base)
target_include_directories(rope_npu_runner PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/..)
target_compile_options(rope_npu_runner PRIVATE
    $<$<COMPILE_LANGUAGE:ASC>:--npu-arch=${SOC_ARCH}>)
